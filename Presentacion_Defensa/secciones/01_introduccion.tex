\section{Introducción}

% --- Slide 3: Hook - "¿Estamos escuchando todas las voces?" ---
\begin{frame}{¿Estamos escuchando todas las voces?}
\begin{columns}[T]

\begin{column}{0.55\textwidth}
\small
Un clasificador aprende \textbf{de lo que ve}. Y la realidad rara vez se reparte por partes iguales:

\vspace{0.3em}
\begin{itemize}\setlength\itemsep{1pt}
  \item \textbf{1} fraude entre \textbf{10.000} transacciones.
  \item \textbf{30} cánceres raros entre \textbf{5.000} radiografías.
  \item \textbf{Un} mensaje de auxilio entre miles de consultas rutinarias.
  \item Discurso de odio en lenguas minorizadas: \textbf{decenas} de ejemplos.
\end{itemize}

\vspace{0.4em}
\begin{insight}
\scriptsize
\textbf{Lo común se escucha. Lo raro se calla.}\\
\alert{¿Cómo mejorar esas clases si apenas tenemos datos?}
\end{insight}
\end{column}

% --- Slide 4: La asimetría — lo poco escuchado es lo que más importa ---
\begin{frame}{Cuando lo poco escuchado es lo que más importa}
\begin{columns}[T]

\begin{column}{0.55\textwidth}
\small
Un modelo puede ser \textbf{correcto en el 95\% de los casos}\\
y, aun así, \textbf{fallar en el 100\% de los que importan}.

\vspace{0.5em}
\textbf{La asimetría duele:}
\begin{itemize}\setlength\itemsep{2pt}
  \item Un cáncer no detectado \alert{$\rightarrow$ diagnóstico tardío}.
  \item Un \emph{phishing} no marcado \alert{$\rightarrow$ una víctima nueva}.
  \item Una crisis no atendida \alert{$\rightarrow$ una persona sola}.
  \item Un fraude que pasa \alert{$\rightarrow$ dinero perdido}.
\end{itemize}

\vspace{0.4em}
\begin{insight}
\scriptsize
La voz que el modelo \textbf{no escuchó}\\
suele cargar con el mayor \textbf{costo humano}.
\end{insight}
\end{column}
