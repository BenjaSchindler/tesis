\setbeamercolor{block title}{fg=white,bg=uaiblue}
\begin{block}{Why it works: cleaner geometry}
\scriptsize
Tighter classes (intra-class distance $0.822$ vs.\ $0.850$) and better separation
(inter-class $0.616$ vs.\ $0.571$), both $p<0.001$.
\end{block}
\end{column}
\begin{column}{0.52\textwidth}
\setbeamercolor{block title}{fg=white,bg=uaigreen}
\begin{block}{Helps most}
\footnotesize
\begin{itemize}\setlength\itemsep{0pt}
  \item \textbf{Hard classes} (F1 $<30\%$): $+10.44$ pp vs.\ $+0.55$ pp for easy ones (F1 $>80\%$).
        \emph{anger}, emotion 10-shot: $27.8\% \to 44.6\%$.
  \item \textbf{Few classes}: best at 6 ($+5.42$ pp, 100\% win rate).
  \item \textbf{10--25 per class}, linear classifier.
\end{itemize}
\end{block}

\setbeamercolor{block title}{fg=white,bg=uaiamber}
\begin{block}{Limitations}
\footnotesize
\begin{itemize}\setlength\itemsep{0pt}
  \item \textbf{Many classes}: $-0.99$ pp at 77, $+0.01$ pp at 150.
  \item \textbf{More data}: only $+0.76$ pp at 50-shot.
  \item \textbf{English only}; multilingual transfer untested.
  \item \textbf{Same encoder} filters and classifies; mitigated by 90.5\% agreement
        across 4 embedding models.
\end{itemize}
\end{block}
\end{column}

% --- Slide 15: Conclusions ---
\begin{frame}[t]{Take-home messages}
\vspace{0.3em}
\begin{columns}[T]
\begin{column}{0.56\textwidth}
\small
\begin{enumerate}\setlength\itemsep{2pt}
  \item \textbf{Distance filtering works:} $+2.61$ pp over SMOTE
        ($d=0.95$, 88.9\% win rate) across 13 datasets.
  \item \textbf{Simple beats complex:} distance alone outperforms LOF, cosine and
        combined criteria. Don't over-filter.
  \item \textbf{Decoupled and general:} robust across 5 LLMs;
        transfers to NER ($+9.26$ pp) unchanged.
\end{enumerate}

\vspace{0.2em}
{\scriptsize\textbf{Next:} multilingual embeddings, local open-source LLMs,
joint tuning of filter and weighting, other tasks (QA, summarization).}
\end{column}
\begin{column}{0.44\textwidth}
\setbeamercolor{block title}{fg=white,bg=uaiorange}
\begin{block}{Recipe}
\footnotesize
\begin{enumerate}\setlength\itemsep{1pt}
  \item Prompt an LLM with your few real examples.
  \item Generate a \textbf{3$\times$ surplus}.
  \item Embed; score by \textbf{distance to the nearest real example}.
  \item Keep the top half; \textbf{soft-weight} them.
  \item Train a linear classifier.
\end{enumerate}
\end{block}
\end{column}
\end{columns}
