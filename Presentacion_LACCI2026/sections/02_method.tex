\begin{frame}{Geometric filters, from simple to complex}
\begin{columns}[c]
\begin{column}{0.58\textwidth}
\footnotesize
\renewcommand{\arraystretch}{1.3}
\begin{tikzpicture}
\node[inner sep=0pt] (tab) {%
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{2.0cm}>{\raggedright\arraybackslash}p{5.3cm}@{}}
\toprule
\textbf{Filter} & \textbf{Criterion} \\
\midrule
None (control) & Random selection from LLM candidates. \\
\rowcolor{uaigreen!14}
\textbf{Cascade L1}\,$\star$ & \textbf{Euclidean distance to real examples of the class.} \\

\bottomrule
\end{tabular}};
\draw[-{Stealth[length=5pt]}, very thick, uaigray!45]
  ([xshift=-6pt, yshift=-16pt]tab.north west) -- ([xshift=-6pt, yshift=2pt]tab.south west)
  node[midway, sloped, below, font=\scriptsize, color=uaigray] {more criteria};
\end{tikzpicture}
\end{column}
\begin{column}{0.42\textwidth}
\small
\textbf{Cascade level 1 score:}
\[
d(x) = \min_{a \in \mathcal{A}_c} \lVert x - a \rVert_2,
\qquad s(x) = 1 - \frac{d(x)}{\max d}
\]
{\scriptsize $\mathcal{A}_c$: real examples of class $c$.}

\vspace{0.4em}
Candidates are \textbf{ranked} and the top-$N$ are kept: no hard threshold to tune.

\vspace{0.6em}
\begin{insight}
\footnotesize
\textbf{Spoiler:} the simplest filter (L1) wins.
\end{insight}
\end{column}
\end{columns}
\end{frame}

% --- Slide 7: Soft weighting ---
\begin{frame}{Soft weighting: beyond accept / reject}
\begin{columns}[c]
\begin{column}{0.5\textwidth}
\small
Instead of a binary decision, each kept candidate gets a \textbf{training weight}:

\begin{enumerate}\setlength\itemsep{1pt}\footnotesize
  \item Compute cascade L1 scores.
  \item Keep the top-$N$ candidates.
  \item Min-max normalize: $s_{\text{norm}}(x) \in [0,1]$.
  \item Temperature scaling:
\end{enumerate}
\vspace{-0.4em}
\[
w(x) = \max\!\left(w_{\min},\ s_{\text{norm}}(x)^{1/T}\right)
\]
{\scriptsize $T = 0.5$ (exponent 2), $w_{\min}=0$. Real samples always get $w=1$.\\
Weights are passed to the classifier as \texttt{sample\_weight}.}
\end{column}
\begin{column}{0.5\textwidth}
\centering
\begin{tikzpicture}
\begin{axis}[
  lacci, width=6.4cm, height=5.0cm,
  xmin=0, xmax=1, ymin=0, ymax=1.05,
  xtick={0,0.25,0.5,0.75,1}, ytick={0,0.25,0.5,0.75,1},
  xticklabels={0,,0.5,,1}, yticklabels={0,,0.5,,1},
  xmajorgrids, ymajorgrids,
  xlabel={normalized score $s_{\text{norm}}(x)$},
  ylabel={training weight $w(x)$},
  legend pos=north west, clip=false,
]
  \addplot[methodSMOTE, thick, dashed, domain=0:1, samples=2] {x};
  \addlegendentry{$T=1$ (linear)}
  \addplot[uaiorange, line width=1.6pt, domain=0:1, samples=60] {x^2};
  \addlegendentry{$T=0.5$ (ours)}
  % worked example: a median candidate gets only a quarter of the weight
  \draw[uaigray, densely dotted] (axis cs:0.5,0) -- (axis cs:0.5,0.25) -- (axis cs:0,0.25);
  \fill[uaiorange] (axis cs:0.5,0.25) circle (2pt);
  \node[font=\scriptsize, color=uaiorange, anchor=north west] at (axis cs:0.52,0.22) {$s=0.5 \Rightarrow w=0.25$};
\end{axis}
\end{tikzpicture}\\[2pt]
{\scriptsize $T<1$ concentrates weight on the most class-consistent candidates.}
\end{column}
\end{columns}

\keymessage{Geometrically consistent samples contribute more to training.}
\end{frame}
